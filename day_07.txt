::q1.

vi lifePath.sh

echo "Enter the date : "
read in_date

a=$(($in_date%10))
b=$(($in_date/10))
c=$(($a+$b))

case $c in
	1) echo "Lucky";;
	2) echo "Carefuly do your work";;
	3) echo "Storger";;
	4) echo "Happy";;
	5) echo "Can get help";;
	6) echo "Doubt";;
	7) echo "Sad";;
	8) echo "Like";;
	9) echo "Courage";;
	*) echo "Unrecognized number";
esac

chmod 777 lifePath.sh

./lifePath.sh

vi operations.sh

sum=0
mul=1
for number in 1 2 3 4 5
		do
			
			sum=$((sum+number))
			
			mul=$((mul*number))
		done
	
	echo "summation = " $sum
	echo "multiplication = " $mul
