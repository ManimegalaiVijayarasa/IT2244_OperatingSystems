::q2)

vi sumOfPrimeNo.sh
	

sum=0

is_prime() {
    local n=$1
    if ((n <= 1)); then
        return 1
    fi
    for ((i=2; i*i<=n; i++)); do
        if ((n % i == 0)); then
            return 1
        fi
    done
    return 0
}

for ((n=1; n<=100; n++)); do
    if is_prime "$n"; then
        sum=$((sum + n))
    fi
done

echo "Sum of prime numbers between 1 and 100 is : $sum"


chmod 777 sumOfPrimeNo.sh

./sumOfPrimeNo.sh
