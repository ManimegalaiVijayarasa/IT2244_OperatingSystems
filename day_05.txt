::View contents of the file, page by page.
more data.csv
less data.csv


::Searches for rows containing the word "Engineering".
grep 'Engineering' data.csv


::Shows the first 10 lines.
head data.csv

::First 5 lines.
head -5 data.csv 

::Last 3 lines.
tail -3 data.csv

::Shows line 7.
awk 'NR==7' data.csv

::Also shows line 7.
head -n7 data.csv | tail -n1

::Shows line 2.
head -n2 data.csv | tail -n1 


::Prints number of fields (columns) in the first line.
awk -F, '{print NF;exit}' data.csv 

::Prints the 3rd column (Age).
awk -F, '{print $3}' data.csv 

::Also prints 3rd column using cut.
cut -d, -f3 data.csv 

::Prints Name and Age.
awk -F, '{print $2, $3}' data.csv

::Appends Name and Age to xyz1.csv.
awk -F, '{print $2, $3}' data.csv >> xyz1.csv


::Sort by Salary (4th column) numerically.
sort -t, -k4 -n data.csv 

::Extracts Salary column.
awk -F, '{print $4}' data.csv

::Prints sorted Salary values.
awk -F, '{print $4}' data.csv | sort 

::Sort by Age (3rd column), numerical reverse.
sort -t',' -k3,3nr data.csv 

::Sort by Salary ascending.
sort -t',' -k4,4n data.csv 

::Sort by Salary descending.
sort -t',' -k4,4nr data.csv 

::Same as above.
sort -t',' -k4,4 -r data.csv

::Sort by Name (2nd col), then Salary descending.
sort -t',' -k2,2 -k4,4 -r data.csv 

::Sort by Department, then Salary descending.
sort -t',' -k5,5 -k4,4 -r data.csv
