Q9.

// c operators 
/*
	+
	-
	*
	/
	% -> remainder

*/

vi nine.c

#include<stdio.h>

int main(){
	int num1;
	printf("Enter number 1 : ");
	scanf("%d", &num1);
	
	int num2;
	printf("Enter number 2 : ");
	scanf("%d", &num2);
	
	int add= num1+num2;
	int sub= num1-num2;
	int mul= num1*num2;
	int div= num1/num2;
	
	printf("\nAddition  is :%d ",add);
	printf("\nSubstraction  is :%d ",sub);
	printf("\nMultiplication  is :%d ",mul);
	printf("\nDivision  is :%d ",div);
	
	
return 0;
}

press Esc
type :wq!


gcc nine.c -o nine    

./nine
