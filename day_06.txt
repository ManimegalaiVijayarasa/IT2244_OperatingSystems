::q3.

vi Q_3.sh

echo "calculator with arithmetic operators."
echo "Enter the first number: "
read num1
echo "Enter the second number: "
read num2

add=$(($num1+$num2))
sub=$(($num1-$num2))
mul=$(($num1*$num2))
div=$(($num1/$num2))


echo "Summation: " $add
echo "Substraction: " $sub
echo "Division: " $div
echo "Multiplication:" $mul

chmod 777 Q_3.sh

./Q_3.sh
